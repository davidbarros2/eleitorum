```text
class EncodingDetectionResult:
    encoding: str          # e.g., "utf-8-sig", "utf-8", "cp1252", "iso-8859-1"

    via_bom: bool          # True if encoding was determined by BOM detection
    raw_chaos: float | None  # underlying charset-normalizer chaos value (None if fallback chain used)

@dataclasses.dataclass(frozen=True)
class ColumnMapping:
    mec_col_index: int | None   # column index of mecanográfico column; None if not detected
    name_col_index: int | None  # column index of name column; None if not detected
    mec_col_label: str | None   # original column label as it appeared in the source header
    name_col_label: str | None  # original column label
    ambiguous_mec_candidates: list[tuple[int, str]]  # (col_index, label) pairs when multiple candidates matched (DET-06)
    ambiguous_name_candidates: list[tuple[int, str]]
    detection_method: Literal["synonym", "format_fallback", "manual"]

# Module-level constants:
MECANOGRAFICO_SYNONYMS: frozenset[str]   # NFKD-normalized synonyms from Eleitorum.md Section 6.5
NAME_SYNONYMS: frozenset[str]
_MEC_FORMAT_PATTERN: re.Pattern           # ^[A-Za-z]{1,2}\d+$ for D-01 fallback
_CONFIDENCE_THRESHOLD: float = 0.85       # INP-07 (D-06)
_CHAOS_THRESHOLD: float = 0.15            # equivalent threshold for charset-normalizer
_ENCODING_FALLBACK_CHAIN: tuple[str, ...] = ("utf-8", "cp1252", "iso-8859-1")
_HEADER_SCAN_WINDOW: int = 10             # first 10 rows for DET-01

def normalize_col_name(s: str) -> str:
    """NFKD normalization for tolerant column matching. Strips diacritics, lowercases, trims.
    Critical: uses NFKD (not NFD) because NFKD decomposes 'º' (U+00BA ordinal indicator) → 'o'."""

def detect_encoding(raw_bytes: bytes) -> EncodingDetectionResult:
    """Detect encoding for CSV/TSV bytes. BOM trusted unconditionally. charset-normalizer used
    when chaos < 0.15 (≈ confidence ≥ 0.85). Falls through to UTF-8 → CP1252 → ISO-8859-1.
    Raises EncodingDetectionError if no encoding decodes the sample cleanly."""

def detect_header_row(rows: list[tuple[Any, ...]]) -> int | None:
    """Score each of the first 10 rows by header-likeness (count of non-empty short text cells +
    bonus for cells matching MECANOGRAFICO_SYNONYMS or NAME_SYNONYMS). Returns the index of the
    highest-scoring row, or None if the maximum score is below threshold (DET-02 manual mode)."""

def detect_columns(
    header_row: tuple[Any, ...],
    data_rows: list[tuple[Any, ...]],
    output_type: Literal["caderno", "elegiveis"],
) -> ColumnMapping:
    """Match header cells to MECANOGRAFICO_SYNONYMS and NAME_SYNONYMS using normalize_col_name.
    If no synonym matches the mec column, scan each column's first 50 data rows for ≥70% match
    against _MEC_FORMAT_PATTERN (D-01 fallback). For output_type='elegiveis', mec_col_index is
    always None (DET-07). Returns ColumnMapping; never raises — None values signal manual-mapping mode."""
```

```text
_MOJIBAKE_PAT = re.compile(r'\xc3[\x80-\xbf]')
def try_fix_mojibake(s):
    if not _MOJIBAKE_PAT.search(s): return s, False
    try:
        fixed = s.encode('latin-1').decode('utf-8')
        if not _MOJIBAKE_PAT.search(fixed): return fixed, True
        return s, False
    except (UnicodeEncodeError, UnicodeDecodeError):
        return s, False
```

```text
from charset_normalizer import from_bytes
def detect_encoding(raw_bytes):
    results = from_bytes(raw_bytes)
    if not results: return _fallback_chain(raw_bytes)
    best = results.best()
    if best is None: return _fallback_chain(raw_bytes)
    if best.bom: return EncodingDetectionResult(best.encoding, 1.0, True, None)
    if best.chaos < 0.15: return EncodingDetectionResult(best.encoding, 1.0 - best.chaos, False, best.chaos)
    return _fallback_chain(raw_bytes)
```

**Pitfalls applicable to this plan:**
- Pitfall 1 — charset-normalizer has no `.encoding_confidence`; use `chaos` and `bom`.
- Pitfall 4 — Case normalization must be batch (after all rows), not per-row.
- Pitfall 5 — Mojibake pattern can misidentify clean 'Ã' text; the round-trip-cleanness check guards against this.
- Pitfall 6 — F/D/B uniqueness must compare uppercase prefixes (handled in validate.py — but transform_mecanografico returns uppercase prefix to make that downstream check straightforward).

**TRF-04 case-normalization batch design:**
The pipeline (plan 05) will:
1. Call `transform_mecanografico` on every row → collect list of TransformResults.
2. Collect the original prefix substrings as separate list (so case-counting is on raw input, not on uppercase output).
3. Call `normalize_mecanografico_case(transforms, raw_prefix_strings)` once after all rows are transformed.
4. The chosen case is applied when writing each row to output.

This is non-negotiable per Pitfall 4 — never apply case normalization row-by-row.

**Whitespace handling (TRF-05/06):**
Use `re.sub(r'\s+', ' ', s.strip())` where `\s` in Python regex matches: ASCII space, tab, newline, carriage return, formfeed, vtab, AND Unicode whitespace including U+00A0 (NBSP) and U+200B (ZWSP). Per RESEARCH.md "Don't Hand-Roll" — never roll a custom whitespace detector.

**Parenthesis stripping (TRF-08):**
`re.sub(r'\s*\([^)]*\)\s*', ' ', s).strip()` — note the surrounding `\s*` swallows whitespace on both sides, then `.strip()` cleans the residue. Re-apply whitespace collapse after.
